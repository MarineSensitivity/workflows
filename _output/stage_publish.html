Published

2026-10-02

Not a pipeline stage — milestone M3 of .claude/plans_todo/2026-10-01 asset store — every distribution file stored once, referenced by every release.md as a dry run. For v9, v8, v7b and v7 it builds, into ~/_big/msens/derived/asset_store/publish_stage/{ver}/, what the release would publish once the store exists: tables/native_asset.parquet (every pointer on the store, with content_hash; v9’s lost vector model rows restored; v7/v7b’s originals and the name-matched rng_iucn ranges), the 512 app shards rebuilt from those tables, and the manifest.json patch. Each is diffed against what is published now (fetched anonymously) to prove that only asset rows/URLs changed. Nothing is written to S3 or any remote. Inputs: stage/object_keys.parquet, stage/assets.parquet, stage/v7_rng_iucn_match_final.parquet from stage_asset_store.qmd. Flags: PS_VERS (default v9,v8,v7b,v7), REDO_PS.

dl <- function(url, dest) {
  if (file_exists(dest) && file_size(dest) > 0 && !redo) return(TRUE)
  dir_create(path_dir(dest)); tmp <- paste0(dest, ".part")
  r <- suppressWarnings(try(utils::download.file(url, tmp, mode = "wb", quiet = TRUE), silent = TRUE))
  if (!inherits(r, "try-error") && identical(r, 0L) && file_exists(tmp) && file_size(tmp) > 0) { file_move(tmp, dest); TRUE } else { unlink(tmp); FALSE }
}

2 1 · v9 and v8: re-point every pointer at the store

Each asset_url (a versioned S3 path, or the file host’s pmtiles/{ver}/…?v=) names an object; object_keys says which store key that object’s pixels live under (a repainted object’s key is the repaint’s). v9 additionally gets its lost vector model rows back (native_asset_restore_model), pointing at the gridded vec_grid COGs, which are the same objects v8 references.

repoint <- function(v) {
  na <- pub_tbl(v, "native_asset"); stopifnot(!is.null(na))
  k  <- ok |> filter(ver == v) |> select(rel, key)
  rel <- rel_of(na$asset_url); key <- k$key[match(rel, k$rel)]
  stopifnot("a pointer names an object the migration map does not know" = !anyNA(key))
  out <- na |> mutate(asset_url = store_url(key), content_hash = hash_of(key))
  if (v == "v9") {
    vg <- ok |> filter(ver == "v9", grp == "vec_grid")
    mu <- tibble(mdl_key = vg$mdl_key, asset_url = store_url(vg$key), content_hash = hash_of(vg$key))
    out <- native_asset_restore_model(as.data.frame(out), as.data.frame(mu)) |> as_tibble()
  }
  stopifnot("duplicate (mdl_key, representation)" = !anyDuplicated(out[c("mdl_key", "representation")]),
            "unexpected columns" = identical(names(out), NA_COLS) || setequal(names(out), NA_COLS))
  list(published = na, staged = out[, NA_COLS])
}

3 2 · v7 and v7b: build native_asset (original next to interpolated)

msens::native_asset_backfill() gives each v1–v7 input that has a published original (the same source object in v8, or v9 where v8 lacks it and the source vintage matches) a native row next to its gridded model row (v7’s own cog/usa05/ COG). IUCN ranges key by scientific NAME in v7 and by id_no in v8+, so they are matched explicitly (key_map): an exact, unambiguous name, whose v7 cells lie on the range (stage_asset_store.qmd §5, cell test). That match is the evidence the vintage check cannot give, so rng_iucn is exempt from the vintage drop.

4 3 · Rebuild the app shards from the staged tables and diff against what is published

app_taxon_shards() / app_alias_shards() are pure functions of the tables. First prove the builder reproduces the published shards from the published tables (the baseline), then rebuild from the staged pointer table and diff.

compare_release <- function(v) {
  t0 <- Sys.time()
  # published-table baseline: the builder must reproduce the published bundle
  c0 <- mk_con(v, "published"); tx0 <- app_taxon_shards(c0, v); al0 <- app_alias_shards(c0, v); dbDisconnect(c0, shutdown = TRUE)
  c1 <- mk_con(v, "staged");    tx1 <- app_taxon_shards(c1, v); al1 <- app_alias_shards(c1, v); dbDisconnect(c1, shutdown = TRUE)
  stopifnot(setequal(names(tx0), names(tx1)))
  out <- glue("{dir_ps}/{v}/app"); dir_create(c(glue("{out}/taxon"), glue("{out}/alias")))
  for (k in names(tx1)) writeLines(app_json(tx1[[k]]), glue("{out}/taxon/{k}.json"))
  for (k in names(al1)) writeLines(app_json(al1[[k]]), glue("{out}/alias/{k}.json"))
  base_tx <- 0L; base_al <- 0L; n_taxa <- 0L; other_diff <- 0L; inputs_changed <- 0L
  merged_url <- 0L; a_pub <- 0L; a_new <- 0L; a_added <- 0L; a_removed <- 0L; a_url <- 0L; a_other <- 0L; both_pub <- 0L; both_new <- 0L; sz <- numeric()
  for (k in names(tx1)) {
    p <- rd_pub(v, "taxon", k); a <- fromJSON(toJSON(tx1[[k]], auto_unbox = TRUE, null = "null", digits = NA), simplifyVector = FALSE)
    if (identical(canon(tx0[[k]]), canon(p))) base_tx <- base_tx + 1L
    sz <- c(sz, gz_bytes(app_json(tx1[[k]])))
    stopifnot(setequal(names(p$taxa), names(a$taxa)))
    for (nm in names(a$taxa)) {
      n_taxa <- n_taxa + 1L
      # both sides are parsed JSON (the staged shard went through one toJSON/fromJSON), so identical() on the lists is exact
      if (!identical(strip_assets(a$taxa[[nm]]), strip_assets(p$taxa[[nm]]))) other_diff <- other_diff + 1L
      if (!identical(a$taxa[[nm]]$inputs, p$taxa[[nm]]$inputs)) inputs_changed <- inputs_changed + 1L
      if (!identical(a$taxa[[nm]]$merged$url, p$taxa[[nm]]$merged$url)) merged_url <- merged_url + 1L
      ap <- assets_of(p$taxa[[nm]]); an <- assets_of(a$taxa[[nm]])
      kp <- vapply(ap, function(x) paste(x$mdl, x$rep, x$type), ""); kn <- vapply(an, function(x) paste(x$mdl, x$rep, x$type), "")
      a_pub <- a_pub + length(ap); a_new <- a_new + length(an)
      a_added <- a_added + sum(!kn %in% kp); a_removed <- a_removed + sum(!kp %in% kn)
      both <- intersect(kp, kn); up <- vapply(ap, `[[`, "", "url"); un <- vapply(an, `[[`, "", "url")
      a_url <- a_url + sum(up[match(both, kp)] != un[match(both, kn)])
      for (i in p$taxa[[nm]]$inputs) both_pub <- both_pub + (length(i$assets) > 1)
      for (i in a$taxa[[nm]]$inputs) both_new <- both_new + (length(i$assets) > 1)
    }
  }

  valid <- all(vapply(names(tx1), function(k) isTRUE(tryCatch({ app_validate(tx1[[k]], "taxon"); TRUE }, error = function(e) FALSE)), NA)) &&

5 4 · The manifest patch

A manifest says how to render a release. v7/v7b have no tables.native_asset and capabilities.native_representation = false (so the app offers no Original|Interpolated toggle even though the shards now carry both). The patch is exactly those two keys for v7/v7b and nothing for v8/v9 (their manifests already list native_asset). app.built_at (the OPFS cache key) and the app.probed block are NOT touched here: they are re-measured by build_app_bundle.qmd AFTER the store exists.

Code 
patch_manifest <- function(v) {
  f <- glue("{dir_pub}/{v}/manifest.json"); stopifnot(dl(glue("{base}/{v}/manifest.json"), f))
  m0 <- fromJSON(f, simplifyVector = FALSE); m <- m0
  if (v %in% c("v7", "v7b")) {
    m$tables$native_asset <- glue("{base}/{v}/tables/native_asset.parquet") |> as.character()
    m$capabilities$native_representation <- TRUE
  }
  dir_create(glue("{dir_ps}/{v}")); write_json(m, glue("{dir_ps}/{v}/manifest.json"), auto_unbox = TRUE, pretty = TRUE, null = "null", digits = NA)
  # the patched manifest must still satisfy the manifest schema (checked only if the PUBLISHED one does: some predate it)
  chk <- function(x) isTRUE(tryCatch({ app_validate(x, "manifest"); TRUE }, error = function(e) FALSE))
  cat(sprintf("%s manifest schema: published %s, patched %s\n", v, chk(m0), chk(m)))
  ch <- function(a, b, path = "") unlist(imap(b, function(x, n) {
    p <- if (nzchar(path)) paste(path, n, sep = ".") else n
    if (is.list(x) && is.list(a[[n]])) ch(a[[n]], x, p) else if (!identical(a[[n]], x)) p else NULL }))
  tibble(ver = v, keys_changed = paste(ch(m0, m), collapse = ", "))
}

## 4 · The manifest patch

```{r}
  m0 <- fromJSON(f, simplifyVector = FALSE); m <- m0
  if (v %in% c("v7", "v7b")) {
    m$tables$native_asset <- glue("{base}/{v}/tables/native_asset.parquet") |> as.character()
    m$capabilities$native_representation <- TRUE
  }
  dir_create(glue("{dir_ps}/{v}")); write_json(m, glue("{dir_ps}/{v}/manifest.json"), auto_unbox = TRUE, pretty = TRUE, null = "null", digits = NA)
  # the patched manifest must still satisfy the manifest schema (checked only if the PUBLISHED one does: some predate it)
  chk <- function(x) isTRUE(tryCatch({ app_validate(x, "manifest"); TRUE }, error = function(e) FALSE))
  cat(sprintf("%s manifest schema: published %s, patched %s\n", v, chk(m0), chk(m)))
  ch <- function(a, b, path = "") unlist(imap(b, function(x, n) {
    p <- if (nzchar(path)) paste(path, n, sep = ".") else n
    if (is.list(x) && is.list(a[[n]])) ch(a[[n]], x, p) else if (!identical(a[[n]], x)) p else NULL }))
  tibble(ver = v, keys_changed = paste(ch(m0, m), collapse = ", "))
}
mf <- map_dfr(vers, patch_manifest); knitr::kable(mf, caption = "manifest.json: keys that differ from the published manifest")
```

## 7 · Backup, then publish (flagged; NOT run in the dry run)

Both chunks are off unless their flag is exactly `1`, and neither runs until Ben says go (`m2-m4-runbook.md`).

**`PS_BACKUP=1`** — read-only. Copies what is published now for each release into `publish_stage/_backup/{ver}/`:
`manifest.json`, `tables/native_asset.parquet` (v7/v7b have none), and the whole `app/` tree (an `aws s3 sync` *from* S3).
With bucket versioning on (runbook step 1) every overwritten object also keeps its previous version for 30 days, so this
is the second safety net, and the only one that lives outside the bucket.

```{r}
#| label: backup
#| eval: !expr identical(Sys.getenv("PS_BACKUP"), "1")

bk <- glue("{dir_ps}/_backup"); dir_create(bk)
for (v in vers) {
  d <- glue("{bk}/{v}"); dir_create(glue("{d}/tables"))
  run <- function(args) { o <- system2("aws", args, stdout = TRUE, stderr = TRUE); st <- attr(o, "status"); if (!is.null(st) && st != 0) stop(paste(utils::tail(o, 3), collapse = "\n")); invisible(o) }
  run(c("s3", "cp", glue("s3://oceanmetrics.io-public/marine-atlas/{v}/manifest.json"), glue("{d}/manifest.json"), "--only-show-errors"))
  has_na <- identical(system2("aws", c("s3api", "head-object", "--bucket", "oceanmetrics.io-public", "--key", glue("marine-atlas/{v}/tables/native_asset.parquet")), stdout = FALSE, stderr = FALSE), 0L)
  if (has_na) run(c("s3", "cp", glue("s3://oceanmetrics.io-public/marine-atlas/{v}/tables/native_asset.parquet"), glue("{d}/tables/native_asset.parquet"), "--only-show-errors"))
  run(c("s3", "sync", glue("s3://oceanmetrics.io-public/marine-atlas/{v}/app/"), glue("{d}/app/"), "--only-show-errors"))
  n <- length(dir_ls(glue("{d}/app"), recurse = TRUE, type = "file"))
  log_info("{v}: backed up manifest.json, native_asset ({has_na}), {n} app objects -> {d}")
}
```

**`PS_PUSH=1`** — uploads, per release, exactly two keys: `{ver}/tables/native_asset.parquet` and `{ver}/manifest.json`
(plain JSON, `no-cache`, as every manifest is). It first runs `scripts/verify_asset_store.R` on the catalog and stops unless
the store matches it, and refuses any key outside that allow-list. The shards are NOT pushed here: `build_app_bundle.qmd`
(`APP_BUNDLE_NATIVE_ASSET_DIR=…/publish_stage APP_BUNDLE_VERS=<ver> APP_BUNDLE_S3=1`) builds them from the staged pointer table once
the store objects exist (its gates HEAD every asset URL) and bumps `built_at`.

```{r}
#| label: push
#| eval: !expr identical(Sys.getenv("PS_PUSH"), "1")

st <- system2("Rscript", c(here("scripts/verify_asset_store.R"), glue("{dir_ps}/assets.parquet")))
if (!identical(st, 0L)) stop("the store does not match the catalog: refusing to publish pointers that name it", call. = FALSE)
allowed <- function(key, v) key %in% c(glue("{v}/tables/native_asset.parquet"), glue("{v}/manifest.json"))
for (v in vers) {
  for (spec in list(list(key = glue("{v}/tables/native_asset.parquet"), file = glue("{dir_ps}/{v}/tables/native_asset.parquet"), args = c("--content-type", "application/octet-stream")),
                    list(key = glue("{v}/manifest.json"), file = glue("{dir_ps}/{v}/manifest.json"), args = c("--content-type", "application/json", "--cache-control", "no-cache")))) {
    stopifnot("refusing a key outside the allow-list" = allowed(spec$key, v), file_exists(spec$file))
    o <- system2("aws", c("s3", "cp", spec$file, glue("s3://oceanmetrics.io-public/marine-atlas/{spec$key}"), spec$args, "--only-show-errors"), stdout = TRUE, stderr = TRUE)
    if (!is.null(attr(o, "status")) && attr(o, "status") != 0) stop(paste(o, collapse = "\n"))
    log_info("pushed {spec$key}")
  }
}
```
