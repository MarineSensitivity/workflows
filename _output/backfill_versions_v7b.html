Published

2026-10-06

Every release v1–v7 exists only as a monolithic sdm.duckdb on the server. To render one in the current app it needs what v8 has: app-facing tables as Parquet, a mdl_seq ↔︎ mdl_key crosswalk, COGs for its surfaces, and a manifest.json describing all of it.

This does that for one release, parameterized by ver, so the same notebook backfills each in turn.

Two halves, deliberately separated. The tables + crosswalk + manifest are cheap (minutes) and are what make a version discoverable. The COG build is expensive — ~30,000 rasters from a 1.18-billion-row surface — and is gated behind BACKFILL_COGS=1. That box also serves the live apps on 4 cores, so the heavy half should be run deliberately, not as a side effect.

1 What makes this affordable

The content-addressed store means a release only pays for surfaces nobody has published yet. Measured across v3/v5/v6/v7: 120,974 model-rows collapse to 19,766 unique surfaces (6.12×), and v6↔︎v7 is 30,061/30,061 identical. So the first usa05 backfill is the expensive one and each subsequent release is mostly hashing.

3 The mdl_seq ↔︎ mdl_key crosswalk

mdl_seq is an autoincrement that renumbers on every rebuild, so a v6 deep link resolves to a different species in v7 — which is why URLs had to move to the stable mdl_key. The old model table happens to carry everything needed to mint one: ds_key plus taxa, which is the source species id (am_0.05 / ITS-Mam-180528). So the crosswalk is mechanical, not hand-built.

Code 
# MERGED models need the taxon's identity, not the model row's `taxa`: v7 stores
# "ms_merge:430645" there, which would mint ms_merge|ms_merge:430645 and match
# nothing. The merged key is ms_merge|{AUTHORITY}:{taxon_id} -- so a v7 merged
# model and a v8 one for the SAME species share a key, which is the whole point.
model <- dbGetQuery(con, "
  SELECT m.mdl_seq, m.ds_key, m.taxa, t.taxon_authority, t.taxon_id
  FROM model m LEFT JOIN taxon t ON t.mdl_seq = m.mdl_seq")

# v1-v7 spell AquaMaps 'am_0.05'; the mdl_key grammar uses 'am'. The rule lives
# in msens::normalize_ds_key() -- it is a rule about the key grammar, and rules
# belong where a test can assert them (test-mdl_key.R round-trips it against the
# real published v1/v3/v7 model_asset rows).

is_merged <- model$ds_key == "ms_merge"
xwalk <- model |>
  mutate(ds_key  = msens::normalize_ds_key(ds_key),
         mdl_key = ifelse(
           is_merged & !is.na(taxon_authority),
           msens::mdl_key_merged(ifelse(is.na(taxon_authority), "WORMS", taxon_authority),
                                 taxon_id),
           msens::mdl_key_raw(ds_key, taxa)))

# a merged model with no taxon row cannot be keyed; report rather than mint a
# plausible-looking key that resolves to nothing
n_orphan <- sum(is_merged & is.na(model$taxon_authority))
if (n_orphan) log_warn("{n_orphan} merged model(s) have no taxon row - keyed from `taxa` instead")
stopifnot("mdl_key collision" = !any(duplicated(xwalk$mdl_key)))

log_info("{nrow(xwalk)} models crosswalked")

built <- 0L
if (do_cogs) {
  dir.create(dir_cog, recursive = TRUE, showWarnings = FALSE)
  grid <- msens::grid_spec_for(grid_id)

  todo <- h |> distinct(content_hash, .keep_all = TRUE) |> filter(!content_hash %in% store)
  if (!is.na(test_n) && test_n > 0) { todo <- head(todo, test_n); log_warn("TEST_N={test_n}") }
  log_info("{nrow(todo)} raster(s) to build")

  for (i in seq_len(nrow(todo))) {
    r   <- todo[i, ]
    out <- file.path(dir_cog, basename(msens::content_key(grid_id, r$content_hash)))
    if (file.exists(out)) next
    d <- dbGetQuery(con, glue("SELECT cell_id, {val} AS val FROM model_cell WHERE mdl_seq = {r$mdl_seq}"))
    if (nrow(d)) {
      msens::publish_cog(d$cell_id, d$val, out, grid,
                         datatype = "INT1U", nodata = 0, overview = FALSE)
      built <- built + 1L
    }
    if (i %% 500 == 0) log_info("  {i}/{nrow(todo)}")
  }
  log_info("built {built} COG(s), {round(sum(file.info(list.files(dir_cog, full.names=TRUE))$size, na.rm=TRUE)/1024^2)} MB")
} else {
  log_info("BACKFILL_COGS unset - skipping the expensive half")

6 Export the app-facing tables

# `release_method` exists only where a release states its own rules as data (v7b onward:
# turtle_suit_min, turtle_fill, coverage_floor_pct, prev). `intersect()` keeps this a no-op
# for every release without it, rather than a missing-table error.
want <- c("cell", "taxon", "model", "metric", "cell_metric", "zone", "zone_cell",
          "zone_metric", "zone_taxon", "dataset", "taxon_model", "listing", "release_method")
have <- intersect(want, dbListTables(con))

sizes <- vapply(have, function(t) {
  d <- dbGetQuery(con, glue("SELECT * FROM {t}"))
  f <- file.path(dir_tbl, paste0(t, ".parquet"))
  arrow::write_parquet(d, f, compression = "zstd")
  nrow(d)
}, numeric(1))

# Whether `taxon_model` is DERIVED here rather than published by that release --
# recorded in the manifest so nobody mistakes reconstructed edges for original.
tm_reconstructed <- FALSE

# Reconstruct `taxon_model` for the releases that predate it (v1/v2).
#
# Those releases record the taxon->model edges WIDE: one integer column per
# dataset on `taxon` (am_0.05, bl, ch_fws, ch_nmfs, rng_fws, +rng_iucn in v2)
# holding that dataset's contributing mdl_seq, plus `mdl_seq` for the merged
# model and `n_ds` for how many datasets contributed. Unpivoting recovers the
# real edges -- this is derived, not invented.
#
# Without it v1/v2 have no taxon->model relation at all: their view DBs carry 11
# views instead of 13, and nothing can answer "which models fed this taxon" for
# the two oldest releases.
if (!"taxon_model" %in% have) {
  tcols <- dbListFields(con, "taxon")
  ds_cols <- intersect(c("am_0.05", "bl", "ch_fws", "ch_nmfs", "rng_fws", "rng_iucn"), tcols)
  if (length(ds_cols) && all(c("mdl_seq", "n_ds") %in% tcols)) {
    sel <- paste(sprintf(
      "SELECT taxon_id, '%s' AS ds_key, \"%s\" AS mdl_seq FROM taxon WHERE \"%s\" IS NOT NULL",
      ds_cols, ds_cols, ds_cols), collapse = " UNION ALL ")
    tm <- dbGetQuery(con, glue(
      "{sel} UNION ALL SELECT taxon_id, 'ms_merge', mdl_seq FROM taxon WHERE mdl_seq IS NOT NULL"))

    # n_ds is recorded independently of the wide columns, so it checks the unpivot:
    # per-dataset edges (excluding the merged row) should equal it.
    #
    # A small shortfall is EXPECTED and is a ceiling of the source layout, not a
    # bug here: one integer column per dataset can hold only ONE model per dataset
    # per taxon, so a taxon with two models from the same dataset is counted twice
    # by n_ds and stored once. Measured on v1: 5 taxa, 8 edges of 17,676 (0.045%),
    # all BirdLife-id taxa carrying only rng_fws. Warn with the number so the
    # magnitude stays visible; do not fail, and do not silently accept a LARGE gap.
    chk <- dbGetQuery(con, glue(
      "SELECT sum(n_ds) AS expected FROM taxon WHERE n_ds IS NOT NULL"))$expected
    got <- sum(tm$ds_key != "ms_merge")
    if (!isTRUE(all.equal(as.numeric(chk), as.numeric(got)))) {
      pct <- 100 * (as.numeric(chk) - got) / as.numeric(chk)
      log_warn("taxon_model reconstruct: {got} per-dataset edges vs n_ds {chk} ",
               "({round(pct, 3)}% short — the wide layout holds one model per dataset per taxon)")
      stopifnot("taxon_model reconstruct is missing >1% of edges" = pct < 1)
    }
    stopifnot("reconstructed taxon_model is empty" = nrow(tm) > 0)

    f_tm <- file.path(dir_tbl, "taxon_model.parquet")
    arrow::write_parquet(tm, f_tm, compression = "zstd")

    # manifest_build() introspects the CONNECTION, never dir_tbl -- deliberately,
    # so a manifest cannot claim a table the release does not have. The cost is
    # that a table written only as a FILE is published but never advertised: the
    # first run of this reconstruction shipped taxon_model.parquet to S3 for v1
    # and v2 (198/208 KB, verified present) while both manifests omitted it, so
    # nothing downstream could discover the table that had just been built.
    # Registering it keeps introspection authoritative and makes it see exactly
    # what was published. A TEMP view is enough (and works on the read-only
    # connection), so the historical release DB is never mutated.
    dbExecute(con, glue(
      "CREATE OR REPLACE TEMP VIEW taxon_model AS SELECT * FROM read_parquet('{f_tm}')"))
    tm_reconstructed <- TRUE

    have <- c(have, "taxon_model")
    sizes <- c(sizes, taxon_model = nrow(tm))
    log_info("taxon_model RECONSTRUCTED for {bver}: {nrow(tm)} edges ",
             "({got} per-dataset + {sum(tm$ds_key == 'ms_merge')} merged) from {length(ds_cols)} wide columns")
  }
}

7 The v8-shaped native_asset (original next to interpolated)

v1-v7 publish one gridded COG per model, which the app used to label native. This reuses the ORIGINAL surface (PMTiles range or the 0.5 deg AquaMaps COG) that a later release already publishes for the same source object, so the atlas can offer Original | Interpolated on this release too. Code only until asked for: gated by BACKFILL_NATIVE=<ref ver> (e.g. v8), it reads the reference’s tables/native_asset.parquet + dataset.parquet (anonymous GET, no write), refuses any dataset whose source vintage differs (msens::native_vintage_check()), and writes native_asset.parquet beside the other tables.

The originals are NOT referenced at the reference release’s versioned paths: they live in the unversioned, content-addressed native store (msens::native_url(), design in publish_native.qmd). BACKFILL_NATIVE_MAP=<parquet> names the asset_url -> hash table from scripts/native_store_map.R (one row per existing v8/v9 object); a matched original with no mapping is an error. Nothing is uploaded here; publishing is the release session’s step.

8 The per-version view DB

The app resolves its database as {dir_big}/serve.duckdb, falling back to the multi-GB sdm.duckdb. Only v8 had a serve.duckdb, so every other release could only be opened by attaching its whole 1.9–3.4 GB database — which is why the app pinned one version at startup instead of switching on ?ver=.

This writes the same KB-sized view DB for this release: views over the Parquet in {ver}/tables/, which the release already syncs to the server. Views are created only for tables that exist — v1/v2 genuinely lack listing and taxon_model, and a view over a missing file fails at CREATE VIEW time, not at query time, so declaring one would break the whole DB rather than degrade one panel.

val AS value is aliased for the v8-era column rename, matching what release_marine-atlas.qmd does, so one app can query either generation without knowing which it is.

f_serve <- file.path(dir_v, "serve.duckdb")
if (file.exists(f_serve)) unlink(f_serve)
sv <- dbConnect(duckdb::duckdb(), f_serve)

# `zone`/`cell_metric`/`zone_metric` carry the scored value; the apps still say `value`
vt   <- c("cell_metric", "zone_metric", "zone")
made <- character()
for (t in unique(c(have, "model_asset", "native_asset"))) {
  f <- file.path(dir_tbl, paste0(t, ".parquet"))
  if (!file.exists(f)) next
  # ask DuckDB what the file holds rather than assuming the generation: v1-v7 name
  # the column `value`, v8 renamed it `val`, and aliasing a column that is already
  # there is a duplicate-name error
  nms  <- names(dbGetQuery(sv, glue("SELECT * FROM read_parquet('{f}') LIMIT 0")))
  cols <- if (t %in% vt && "val" %in% nms && !"value" %in% nms) "*, val AS value" else "*"
  dbExecute(sv, glue("CREATE OR REPLACE VIEW {t} AS SELECT {cols} FROM read_parquet('{f}')"))
  made <- c(made, t)
}

# cell_model is a DIRECTORY of spatially-tiled Parquet beside the tables, not a
# tables/*.parquet file, so it needs its own view. Its presence is exactly what
# manifest_build() keys `cell_species_list` on -- without the view a release
# advertises FALSE and a clicked cell cannot list species, which is the state
# v1-v6 were in until build_v7_cell_model was run for them.
dir_cm <- file.path(dir_v, "cell_model")
if (dir.exists(dir_cm) && length(Sys.glob(file.path(dir_cm, "*", "*.parquet")))) {
  dbExecute(sv, glue(
    "CREATE OR REPLACE VIEW cell_model AS ",
    "SELECT * FROM read_parquet('{dir_cm}/*/*.parquet', hive_partitioning = true)"))
  made <- c(made, "cell_model")
  log_info("cell_model view added ({length(Sys.glob(file.path(dir_cm, '*')))} tiles)")
}

# RECORD THE GRID, then prove the stored tile ids agree with it.
#
# cell_model is partitioned by a tile id computed from the grid WIDTH, and msens resolved
# that width from a `cell_grid` table that no release had ever written -- so it always fell
# back to global05's 7200. On every usa05 release that computes a different, perfectly VALID
# tile id, so `WHERE tile IN (...)` pruned away the only partition holding the cells and the

# right. cell_model_tile_check() is the check that cannot pass on a mismatch -- it recomputes

msens::cell_grid_write(sv, grid_id)

  log_info("cell_model tile ids verified against grid {grid_id} (ncol {msens::cell_grid_ncol(sv)})")
}
made <- c(made, "cell_grid")

10 Publish to S3

Code 
# The rstudio CONTAINER has no `aws` (the host does). Adding it to
# server/rstudio/Dockerfile is the reproducible fix, but it only takes effect on
# the next container recreate -- and a recreate resets that container's R
# library, taking the live Shiny apps down until they are reinstalled. So when
# the CLI is absent this stages locally and prints the exact host-side command,
# rather than failing with `system2: Function not implemented`.
# presence of the CLI is NOT the question -- whether it can AUTHENTICATE is.
# The container gained `aws` before it gained credentials, and a credential-less
# CLI fails at the publish step AFTER the expensive build, which is the worst
# place to find out.
has_aws <- nzchar(Sys.which("aws")) &&
  identical(0L, suppressWarnings(system2("aws", c("sts", "get-caller-identity"),
                                         stdout = FALSE, stderr = FALSE)))
do_s3   <- Sys.getenv("BACKFILL_NO_S3") == "" && has_aws
if (!has_aws)
  log_warn(paste0("no `aws` in this container - staged locally. Publish from the HOST:\n",
                  "  aws s3 sync {dir_tbl} {s3_atlas}/{bver}/tables\n",
                  "  aws s3 cp {f_mfst} {s3_atlas}/{bver}/manifest.json ",
                  "--content-type application/json --cache-control no-cache"))
if (do_s3) {
  s3_ver <- glue("{s3_atlas}/{bver}")
  push <- function(src, dst, ct = NULL) {
    a <- c("s3", "sync", shQuote(src), shQuote(dst),
           if (!is.null(ct)) c("--content-type", shQuote(ct)),
           "--only-show-errors", "--no-progress")
    o <- system2("aws", a, stdout = TRUE, stderr = TRUE)
    if (!is.null(attr(o, "status")) && attr(o, "status") != 0)
      stop(glue("aws s3 sync {src} failed: "), paste(o, collapse = "\n"))
  }
  push(dir_tbl, glue("{s3_ver}/tables"))
  # the manifest is a MUTABLE pointer: a cached copy advertises capabilities the
  # release no longer has
  system2("aws", c("s3", "cp", shQuote(f_mfst), shQuote(glue("{s3_ver}/manifest.json")),
                   "--content-type", shQuote("application/json"),
                   "--cache-control", shQuote("no-cache"), "--only-show-errors"))
  if (do_cogs && dir.exists(dir_cog))
    push(dir_cog, glue("{s3_atlas}/cog/{grid_id}"))

  # verify by READING BACK through the public URL, as the app would
  got <- msens::atlas_manifest(bver, refresh = TRUE)
  stopifnot("published manifest does not round-trip" = identical(got$ver, bver))
  log_info("verified {msens::atlas_base_url()}/{bver}/manifest.json")
} else {
  log_info("BACKFILL_NO_S3 set - staged locally only")
}

11 STAC + app reload: what “a release landed” means

Publishing the tables and the manifest is not the end of a release. Two things used to be left to memory, and v7b (v7.1) showed both: it was absent from the STAC catalog (no committed step registered a legacy release, and msens::stac_build() could not build one), and the preview Shiny instance answered HTTP 500 for it until someone happened to reload the apps, because a worker memoizes the registry and manifests it read before the data existed. Both now happen here, on every run that publishes — BACKFILL_NO_STAC=1 / BACKFILL_NO_RELOAD=1 opt out.

A third thing follows the STAC registration: the alias of the root catalog on the main site (marinesensitivity.org/stac/catalog.json, repo MarineSensitivity.github.io) is compared with the root just deployed (libs/stac_alias.R). A lagging alias is a WARN naming the missing children — it legitimately lags at this moment — and STAC_ALIAS_PUSH=1 writes, commits and pushes it, then waits for GitHub Pages to serve it.

Rendered on the server (scripts/srv_render.sh), the catalog directory is the deployed one (/share/data/derived/stac, served at file.marinesensitivity.org/stac), so there is nothing to rsync; on a laptop the tree is staged beside the release and not deployed.

Every release v1–v7 exists only as a monolithic `sdm.duckdb` on the server. To render one in the
current app it needs what v8 has: **app-facing tables as Parquet**, a **`mdl_seq` ↔ `mdl_key`
crosswalk**, **COGs** for its surfaces, and a **`manifest.json`** describing all of it.

This does that for one release, parameterized by `ver`, so the same notebook backfills each in turn.

> **Two halves, deliberately separated.** The tables + crosswalk + manifest are cheap (minutes) and
> are what make a version *discoverable*. The COG build is expensive — ~30,000 rasters from a
> 1.18-billion-row surface — and is gated behind `BACKFILL_COGS=1`. That box also serves the live
> apps on 4 cores, so the heavy half should be run deliberately, not as a side effect.

## What makes this affordable

The content-addressed store means a release only pays for surfaces nobody has published yet.
Measured across v3/v5/v6/v7: 120,974 model-rows collapse to **19,766 unique surfaces (6.12×)**, and
v6↔v7 is **30,061/30,061 identical**. So the first `usa05` backfill is the expensive one and each
subsequent release is mostly hashing.

## The `mdl_seq` ↔ `mdl_key` crosswalk

`mdl_seq` is an autoincrement that **renumbers on every rebuild**, so a v6 deep link resolves to a
different species in v7 — which is why URLs had to move to the stable `mdl_key`. The old `model`
table happens to carry everything needed to mint one: `ds_key` plus `taxa`, which *is* the source
species id (`am_0.05` / `ITS-Mam-180528`). So the crosswalk is mechanical, not hand-built.

```{r}
#| label: crosswalk

# MERGED models need the taxon's identity, not the model row's `taxa`: v7 stores
# "ms_merge:430645" there, which would mint ms_merge|ms_merge:430645 and match
# nothing. The merged key is ms_merge|{AUTHORITY}:{taxon_id} -- so a v7 merged
# model and a v8 one for the SAME species share a key, which is the whole point.
model <- dbGetQuery(con, "
  SELECT m.mdl_seq, m.ds_key, m.taxa, t.taxon_authority, t.taxon_id
  FROM model m LEFT JOIN taxon t ON t.mdl_seq = m.mdl_seq")

# v1-v7 spell AquaMaps 'am_0.05'; the mdl_key grammar uses 'am'. The rule lives
# in msens::normalize_ds_key() -- it is a rule about the key grammar, and rules
# belong where a test can assert them (test-mdl_key.R round-trips it against the
# real published v1/v3/v7 model_asset rows).

is_merged <- model$ds_key == "ms_merge"
xwalk <- model |>
  mutate(ds_key  = msens::normalize_ds_key(ds_key),
         mdl_key = ifelse(
           is_merged & !is.na(taxon_authority),
           msens::mdl_key_merged(ifelse(is.na(taxon_authority), "WORMS", taxon_authority),
                                 taxon_id),
           msens::mdl_key_raw(ds_key, taxa)))

# a merged model with no taxon row cannot be keyed; report rather than mint a
# plausible-looking key that resolves to nothing
n_orphan <- sum(is_merged & is.na(model$taxon_authority))
if (n_orphan) log_warn("{n_orphan} merged model(s) have no taxon row - keyed from `taxa` instead")
stopifnot("mdl_key collision" = !any(duplicated(xwalk$mdl_key)))

log_info("{nrow(xwalk)} models crosswalked")
knitr::kable(head(xwalk, 5))
```

```{r}
# Whether `taxon_model` is DERIVED here rather than published by that release --
# recorded in the manifest so nobody mistakes reconstructed edges for original.
tm_reconstructed <- FALSE

# Reconstruct `taxon_model` for the releases that predate it (v1/v2).
#
# Those releases record the taxon->model edges WIDE: one integer column per
# dataset on `taxon` (am_0.05, bl, ch_fws, ch_nmfs, rng_fws, +rng_iucn in v2)
# holding that dataset's contributing mdl_seq, plus `mdl_seq` for the merged
# model and `n_ds` for how many datasets contributed. Unpivoting recovers the
# real edges -- this is derived, not invented.
#
# Without it v1/v2 have no taxon->model relation at all: their view DBs carry 11
# views instead of 13, and nothing can answer "which models fed this taxon" for
# the two oldest releases.
if (!"taxon_model" %in% have) {
  tcols <- dbListFields(con, "taxon")
  ds_cols <- intersect(c("am_0.05", "bl", "ch_fws", "ch_nmfs", "rng_fws", "rng_iucn"), tcols)
  if (length(ds_cols) && all(c("mdl_seq", "n_ds") %in% tcols)) {
    sel <- paste(sprintf(
      "SELECT taxon_id, '%s' AS ds_key, \"%s\" AS mdl_seq FROM taxon WHERE \"%s\" IS NOT NULL",
      ds_cols, ds_cols, ds_cols), collapse = " UNION ALL ")
    tm <- dbGetQuery(con, glue(
      "{sel} UNION ALL SELECT taxon_id, 'ms_merge', mdl_seq FROM taxon WHERE mdl_seq IS NOT NULL"))

    # n_ds is recorded independently of the wide columns, so it checks the unpivot:
    # per-dataset edges (excluding the merged row) should equal it.
    #
    # A small shortfall is EXPECTED and is a ceiling of the source layout, not a
    # bug here: one integer column per dataset can hold only ONE model per dataset
    # per taxon, so a taxon with two models from the same dataset is counted twice
    # by n_ds and stored once. Measured on v1: 5 taxa, 8 edges of 17,676 (0.045%),
    # all BirdLife-id taxa carrying only rng_fws. Warn with the number so the
    # magnitude stays visible; do not fail, and do not silently accept a LARGE gap.
    chk <- dbGetQuery(con, glue(
      "SELECT sum(n_ds) AS expected FROM taxon WHERE n_ds IS NOT NULL"))$expected
    got <- sum(tm$ds_key != "ms_merge")
    if (!isTRUE(all.equal(as.numeric(chk), as.numeric(got)))) {
      pct <- 100 * (as.numeric(chk) - got) / as.numeric(chk)
      log_warn("taxon_model reconstruct: {got} per-dataset edges vs n_ds {chk} ",
               "({round(pct, 3)}% short — the wide layout holds one model per dataset per taxon)")
      stopifnot("taxon_model reconstruct is missing >1% of edges" = pct < 1)
    }
    stopifnot("reconstructed taxon_model is empty" = nrow(tm) > 0)

    f_tm <- file.path(dir_tbl, "taxon_model.parquet")
    arrow::write_parquet(tm, f_tm, compression = "zstd")

    # manifest_build() introspects the CONNECTION, never dir_tbl -- deliberately,
    # so a manifest cannot claim a table the release does not have. The cost is
    # that a table written only as a FILE is published but never advertised: the
    # first run of this reconstruction shipped taxon_model.parquet to S3 for v1
    # and v2 (198/208 KB, verified present) while both manifests omitted it, so
    # nothing downstream could discover the table that had just been built.
    # Registering it keeps introspection authoritative and makes it see exactly
    # what was published. A TEMP view is enough (and works on the read-only
    # connection), so the historical release DB is never mutated.
    dbExecute(con, glue(
      "CREATE OR REPLACE TEMP VIEW taxon_model AS SELECT * FROM read_parquet('{f_tm}')"))
    tm_reconstructed <- TRUE

    have <- c(have, "taxon_model")
    sizes <- c(sizes, taxon_model = nrow(tm))
    log_info("taxon_model RECONSTRUCTED for {bver}: {nrow(tm)} edges ",
             "({got} per-dataset + {sum(tm$ds_key == 'ms_merge')} merged) from {length(ds_cols)} wide columns")
  }
}

# the crosswalk ships as a table of its own: it is what lets a legacy
```

## The v8-shaped `native_asset` (original next to interpolated)

v1-v7 publish one gridded COG per model, which the app used to label `native`. This reuses the
ORIGINAL surface (PMTiles range or the 0.5 deg AquaMaps COG) that a later release already

## The per-version view DB

The app resolves its database as `{dir_big}/serve.duckdb`, falling back to the multi-GB
`sdm.duckdb`. Only v8 had a `serve.duckdb`, so every other release could only be opened by
attaching its whole 1.9–3.4 GB database — which is why the app pinned one version at startup
instead of switching on `?ver=`.

This writes the same **KB-sized view DB** for this release: views over the Parquet in
`{ver}/tables/`, which the release already syncs to the server. Views are created **only for
tables that exist** — v1/v2 genuinely lack `listing` and `taxon_model`, and a view over a missing
file fails at `CREATE VIEW` time, not at query time, so declaring one would break the whole DB
rather than degrade one panel.

`val AS value` is aliased for the v8-era column rename, matching what `release_marine-atlas.qmd`
does, so one app can query either generation without knowing which it is.

```{r}
#| label: viewdb

f_serve <- file.path(dir_v, "serve.duckdb")
if (file.exists(f_serve)) unlink(f_serve)
sv <- dbConnect(duckdb::duckdb(), f_serve)

# `zone`/`cell_metric`/`zone_metric` carry the scored value; the apps still say `value`
vt   <- c("cell_metric", "zone_metric", "zone")
made <- character()
for (t in unique(c(have, "model_asset", "native_asset"))) {
  f <- file.path(dir_tbl, paste0(t, ".parquet"))
  if (!file.exists(f)) next
  # ask DuckDB what the file holds rather than assuming the generation: v1-v7 name
  # the column `value`, v8 renamed it `val`, and aliasing a column that is already
  # there is a duplicate-name error
  nms  <- names(dbGetQuery(sv, glue("SELECT * FROM read_parquet('{f}') LIMIT 0")))
  cols <- if (t %in% vt && "val" %in% nms && !"value" %in% nms) "*, val AS value" else "*"
  dbExecute(sv, glue("CREATE OR REPLACE VIEW {t} AS SELECT {cols} FROM read_parquet('{f}')"))
  made <- c(made, t)
}

# cell_model is a DIRECTORY of spatially-tiled Parquet beside the tables, not a
# tables/*.parquet file, so it needs its own view. Its presence is exactly what
# manifest_build() keys `cell_species_list` on -- without the view a release
# advertises FALSE and a clicked cell cannot list species, which is the state
# v1-v6 were in until build_v7_cell_model was run for them.
dir_cm <- file.path(dir_v, "cell_model")
if (dir.exists(dir_cm) && length(Sys.glob(file.path(dir_cm, "*", "*.parquet")))) {
  dbExecute(sv, glue(
    "CREATE OR REPLACE VIEW cell_model AS ",
    "SELECT * FROM read_parquet('{dir_cm}/*/*.parquet', hive_partitioning = true)"))
  made <- c(made, "cell_model")
  log_info("cell_model view added ({length(Sys.glob(file.path(dir_cm, '*')))} tiles)")
}

# RECORD THE GRID, then prove the stored tile ids agree with it.
#
# cell_model is partitioned by a tile id computed from the grid WIDTH, and msens resolved
# that width from a `cell_grid` table that no release had ever written -- so it always fell
# back to global05's 7200. On every usa05 release that computes a different, perfectly VALID
# tile id, so `WHERE tile IN (...)` pruned away the only partition holding the cells and the
# clicked-cell species list came back EMPTY instead of failing: a v7 cell with 477 models
# reported 0 species, on the promoted release. v8 passed only because 7200 happened to be
# right. cell_model_tile_check() is the check that cannot pass on a mismatch -- it recomputes
# the tile for cells taken from cell_model itself and compares with what is stored.
msens::cell_grid_write(sv, grid_id)
if ("cell_model" %in% made) {
  msens::cell_model_tile_check(sv)
  log_info("cell_model tile ids verified against grid {grid_id} (ncol {msens::cell_grid_ncol(sv)})")
}
made <- c(made, "cell_grid")

# prove each view actually reads, so a missing/corrupt Parquet fails HERE rather than
# surfacing later as an empty panel in the app
counts <- vapply(made, function(t)
  dbGetQuery(sv, glue("SELECT count(*) n FROM {t}"))$n, numeric(1))
dbDisconnect(sv, shutdown = TRUE)
stopifnot("every view must return rows" = all(counts > 0))

log_info("{bver} serve.duckdb: {length(made)} views, {round(file.size(f_serve)/1024)} KB")
```

```{r}
         "{if (is.null(overlays)) 0 else nrow(overlays)} overlay")

# Layer PRESENTATION metadata (display order, category heading, short label) was
# a per-version `layers_{ver}.csv` sitting beside the release on the server, so
# the app needed a local file to know how to label its own layer picker -- and
# v1/v2 have no such file at all, which is one reason they could not render.
# Fold it into the manifest, which is the contract for everything else already.
# NB: layers csv lives under derived/{ver}, NOT big/{ver} where the release db is
f_lyrs <- file.path(path.expand(glue("{dir_derived}/{bver}")), glue("layers_{bver}.csv"))
lyr_meta <- if (file.exists(f_lyrs)) {
  readr::read_csv(f_lyrs, show_col_types = FALSE) |>
    dplyr::transmute(metric_key = lyr, lyr_order = order, category, label = layer)
} else {
  # v1/v2: derive a usable default rather than shipping a manifest the app can
  # only half-read. Order follows metric_seq (the order the release itself
  # defines), category groups by the metric_key stem, label falls back to the
  # description -- all visible as `derived` so the app can say so.
  NULL
}
if (!is.null(scog) && !is.null(lyr_meta))
  scog <- dplyr::left_join(scog, lyr_meta, by = "metric_key")
log_info("layer metadata: {if (is.null(lyr_meta)) 'derived (no layers csv)' else paste(nrow(lyr_meta), 'rows from layers csv')}")

# Zone tiles by vintage, exactly as build_version_manifest.qmd does for v8. This
# notebook omitted them, so every historical manifest published with no zone
# outlines while reporting success -- the app could draw zones on v8 alone.
# `zone_sets` is what lets a release whose own `zone` table predates the
# zone_set_key column resolve its vintages at all.
zsets      <- readr::read_csv(here::here("data/zone_sets.csv"), show_col_types = FALSE)
zone_tiles <- setNames(
  as.list(glue("{msens::atlas_base_url()}/zones/{zsets$zone_set_key}/zones.pmtiles")),
  zsets$zone_set_key)

mfst <- msens::manifest_build(
  con, bver,
  status     = v_row$status,
  base       = msens::atlas_base_url(),
  metrics    = scog,
  zone_tiles = zone_tiles,
  zone_sets  = zsets,
  # COGs are declared only once they exist; until then the app must not offer a
  # layer it cannot draw
  capabilities = list(model_cogs = built > 0 || length(store) > 0,
                      # provenance, not a feature: v1/v2 taxon->model edges are
                      # unpivoted from that release's wide `taxon` columns, not
                      # published by it. Boolean so `manifest_can()`'s isTRUE()
                      # reads it like any other capability.
                      taxon_model_reconstructed = tm_reconstructed),
  # the rules this release used, carried from its own `release_method` table so the docs
  # state them from the data. Absent for every release that does not declare them, and
  # `doc_manifest()$methods` is then NULL -- which is what gates the method prose.
  extra = c(list(title = v_row$title, released = as.character(v_row$released),
                 overlays = overlays,
                 models = glue("{msens::atlas_base_url()}/{bver}/tables/model_asset.parquet")),
            if ("release_method" %in% have)
              list(methods = dbGetQuery(con, "SELECT * FROM release_method"))))

f_mfst <- file.path(dir_v, "manifest.json")
writeLines(as.character(jsonlite::toJSON(mfst, auto_unbox = TRUE, pretty = TRUE, na = "null")), f_mfst)
msens::validate_manifest(jsonlite::fromJSON(f_mfst), ver = bver)

cat(glue("{bver}: id_field={mfst$id_field} grid={mfst$grid_id} tables={length(mfst$tables)} ",
         "metrics={nrow(mfst$metrics)} zones={nrow(mfst$zones)}\n"))
```

## Publish to S3

```{r}
#| label: publish

# The rstudio CONTAINER has no `aws` (the host does). Adding it to
# server/rstudio/Dockerfile is the reproducible fix, but it only takes effect on
# the next container recreate -- and a recreate resets that container's R
# library, taking the live Shiny apps down until they are reinstalled. So when
# the CLI is absent this stages locally and prints the exact host-side command,
# rather than failing with `system2: Function not implemented`.
# presence of the CLI is NOT the question -- whether it can AUTHENTICATE is.
# The container gained `aws` before it gained credentials, and a credential-less
# CLI fails at the publish step AFTER the expensive build, which is the worst
# place to find out.
has_aws <- nzchar(Sys.which("aws")) &&
  identical(0L, suppressWarnings(system2("aws", c("sts", "get-caller-identity"),
                                         stdout = FALSE, stderr = FALSE)))
do_s3   <- Sys.getenv("BACKFILL_NO_S3") == "" && has_aws
if (!has_aws)
  log_warn(paste0("no `aws` in this container - staged locally. Publish from the HOST:\n",
                  "  aws s3 sync {dir_tbl} {s3_atlas}/{bver}/tables\n",
                  "  aws s3 cp {f_mfst} {s3_atlas}/{bver}/manifest.json ",
                  "--content-type application/json --cache-control no-cache"))
if (do_s3) {
  s3_ver <- glue("{s3_atlas}/{bver}")
  push <- function(src, dst, ct = NULL) {
    a <- c("s3", "sync", shQuote(src), shQuote(dst),
           if (!is.null(ct)) c("--content-type", shQuote(ct)),
           "--only-show-errors", "--no-progress")
    o <- system2("aws", a, stdout = TRUE, stderr = TRUE)
    if (!is.null(attr(o, "status")) && attr(o, "status") != 0)
      stop(glue("aws s3 sync {src} failed: "), paste(o, collapse = "\n"))
  }
  push(dir_tbl, glue("{s3_ver}/tables"))
  # the manifest is a MUTABLE pointer: a cached copy advertises capabilities the
  # release no longer has
  system2("aws", c("s3", "cp", shQuote(f_mfst), shQuote(glue("{s3_ver}/manifest.json")),
                   "--content-type", shQuote("application/json"),
                   "--cache-control", shQuote("no-cache"), "--only-show-errors"))
  if (do_cogs && dir.exists(dir_cog))
    push(dir_cog, glue("{s3_atlas}/cog/{grid_id}"))

  # verify by READING BACK through the public URL, as the app would
  got <- msens::atlas_manifest(bver, refresh = TRUE)
  stopifnot("published manifest does not round-trip" = identical(got$ver, bver))
  log_info("verified {msens::atlas_base_url()}/{bver}/manifest.json")
} else {
  log_info("BACKFILL_NO_S3 set - staged locally only")
}
```

## STAC + app reload: what "a release landed" means

Publishing the tables and the manifest is not the end of a release. Two things used to be left to
memory, and v7b (v7.1) showed both: it was **absent from the STAC catalog** (no committed step
registered a legacy release, and `msens::stac_build()` could not build one), and the **preview
Shiny instance answered HTTP 500** for it until someone happened to reload the apps, because a
worker memoizes the registry and manifests it read before the data existed. Both now happen here,
on every run that publishes — `BACKFILL_NO_STAC=1` / `BACKFILL_NO_RELOAD=1` opt out.

A third thing follows the STAC registration: the **alias of the root catalog on the main site**
(`marinesensitivity.org/stac/catalog.json`, repo `MarineSensitivity.github.io`) is compared with the
root just deployed (`libs/stac_alias.R`). A lagging alias is a WARN naming the missing children —
it legitimately lags at this moment — and `STAC_ALIAS_PUSH=1` writes, commits and pushes it, then
waits for GitHub Pages to serve it.

Rendered on the server (`scripts/srv_render.sh`), the catalog directory *is* the deployed one
(`/share/data/derived/stac`, served at `file.marinesensitivity.org/stac`), so there is nothing to
rsync; on a laptop the tree is staged beside the release and not deployed.

```{r}
#| label: stac

source(here::here("libs/app_reload.R"))
on_server <- dir.exists("/share/data/derived/stac")
do_stac   <- Sys.getenv("BACKFILL_NO_STAC")   == ""
do_reload <- Sys.getenv("BACKFILL_NO_RELOAD") == "" && on_server && do_s3

if (do_stac) {
  stac_root <- if (on_server) "/share/data/derived/stac" else file.path(dir_v, "stac")
  tmp       <- tempfile("stac_")
  # a legacy release's local database has no native_asset table even when the pointer table is
  # published (v7, v7b since the asset-store migration), so ask the bucket, as a client would
  u_na   <- glue("{msens::atlas_bases()$atlas}/{bver}/tables/native_asset.parquet")
  has_na <- tryCatch(
    httr2::request(u_na) |> httr2::req_method("HEAD") |> httr2::req_error(is_error = \(r) FALSE) |>
      httr2::req_perform() |> httr2::resp_status() == 200L,
    error = function(e) FALSE)
  log_info("STAC: {bver} publishes tables/native_asset.parquet: {has_na}")
  msens::stac_build(bver, dir_out = tmp, con = con, model_asset = model_asset, has_native_asset = has_na)
  # publish gate (libs/url_gate.R): no bulk file (.tif/.pmtiles/...) of the tree just built may be
  # fetched from a VM host -- checked BEFORE it is copied into the deployed root. URL_AUDIT_ALLOW = escape hatch
  source(here::here("libs/url_gate.R"))
  tree_urls <- unlist(lapply(
    list.files(file.path(tmp, bver), "\\.json$", recursive = TRUE, full.names = TRUE),
    function(f) { txt <- paste(readLines(f, warn = FALSE), collapse = "\n")
                  regmatches(txt, gregexpr("https?://[^\"\\\\ ]+", txt))[[1]] }))
  url_gate(unique(tree_urls), glue("{bver} STAC tree"))
  # replace this version's tree whole (a dropped dataset must not leave a stale node), then
  # REGISTER it in the deployed root: stac_build()'s own root knows only this one version
  dir.create(stac_root, recursive = TRUE, showWarnings = FALSE)
  unlink(file.path(stac_root, bver), recursive = TRUE)
  ok <- file.copy(file.path(tmp, bver), stac_root, recursive = TRUE)
  stopifnot("could not write the STAC tree" = ok)
  kids <- msens::stac_catalog_register(file.path(stac_root, "catalog.json"), bver)
  n_ds <- length(list.dirs(file.path(stac_root, bver), recursive = FALSE))
  # computed OUTSIDE the log string: glue parses what is inside {braces} as R, and a regex's
  # backslashes there are "an unrecognized escape" -- which stopped the first server run here
  kid_vers <- paste(basename(dirname(kids)), collapse = " ")
  log_info("STAC: {bver} written to {stac_root} ({n_ds} datasets); catalog children: {kid_vers}")

  if (on_server) {
    # read it back through the PUBLIC host, as a STAC client would
    u   <- glue("https://file.marinesensitivity.org/stac/{bver}/collection.json")
    got <- tryCatch(jsonlite::fromJSON(u, simplifyVector = FALSE), error = function(e) NULL)
    cat0 <- tryCatch(jsonlite::fromJSON("https://file.marinesensitivity.org/stac/catalog.json",
                                        simplifyVector = FALSE), error = function(e) NULL)
    hrefs <- unlist(lapply(cat0$links, function(l) l$href))
    stopifnot(
      "the published version collection is not readable" = identical(got$id, paste0("msens-", bver)),
      "the published root catalog does not list this version" =
        glue("./{bver}/collection.json") %in% hrefs)
    log_info("verified {u} and its entry in the root catalog")

    # the root was just (re)deployed: compare the main site's alias of it
    # (marinesensitivity.org/stac) -- WARNs while it lags, STAC_ALIAS_PUSH=1 publishes it
    source(here::here("libs/stac_alias.R"))
    stac_alias_sync()
  }
} else log_info("BACKFILL_NO_STAC set - STAC left as it is")

dbDisconnect(con, shutdown = TRUE)

if (do_reload) {
  rc <- system2("bash", c("-c", shQuote(app_reload_cmd())))
  stopifnot("could not touch the apps' restart.txt (public + preview)" = identical(as.integer(rc), 0L))
  log_info("apps reloaded on BOTH instances (public + preview): {bver}'s data is what they now read")
} else log_info("app reload skipped (not on the server, nothing published, or BACKFILL_NO_RELOAD)")
```
