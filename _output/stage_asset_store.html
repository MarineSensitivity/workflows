2 1 · The key is the hash of the PIXELS the file holds

M1 keyed the raster families by the hash of the raw source rows. That cannot tell trunc from round (v8’s 192 suitability-only merged models were painted with trunc(max), v9’s with round(max): the same rows, different files, one key) and gives two models whose rows differ below the quantum different keys for identical pixels. msens::pixel_hashes() (0.47.0) hashes the values as the writer stores them — trunc into INT1U (verified: every pixel of an AquaMaps model equals trunc(val); 54% equal round(val)), sub-1 pixels dropped as NoData — which equals content_hashes_df(native_raster_rows(<the painted COG>)). The check “does this object hold the rows its key says” is then one equality, and the key is still known before any file is built.

4 3 · Why the failures fail

For every inconsistent object: the cell-level difference against the current rows (cells only in the COG, cells only in the rows, shared cells with another value), whether the rows repeat a cell, and whether the COG is another model’s current surface (the positional-mid bug in publish_native.qmd painted 1,054 COGs from another species’ partition). A small edge failure differs in < 1% of its pixels; a gross one in ≥ 1%.

# the rows that SHOULD be in the object, per folder -- a self-contained worker: DuckDB reads the model's own rows (max(val) per cell,
# the value the merge consumes), the decoded COG is registered as a table, and ONE full outer join counts the differences
# (the base-R merge() version took 2-68 s per object; this takes 0.2-2 s and returns identical numbers, checked on 8 objects)
diag_worker <- function(o, atlas_root, grid) {
  suppressMessages({library(DBI); library(duckdb); library(terra); library(msens)})
  con <- dbConnect(duckdb(), config = list(threads = "2", memory_limit = "3GB")); on.exit(dbDisconnect(con, shutdown = TRUE))
  ad <- file.path(atlas_root, o$ver, "marine-atlas")
  rows_sql <- if (o$grp %in% c("am", "ax", "dps_nmfs", "vec_grid")) {
    ds <- sub("\\|.*$", "", o$mdl_key); id <- sub("^[^|]*\\|", "", o$mdl_key)
    dd <- file.path(ad, "dist", paste0("dataset=", ds))
    # a model can span several files (rng_turtle_swot_dps: 14 files, 6 models; vec_grid painted ONE file per model): for a small
    # dataset read the model's rows across ALL of them; for the big per-model ones the file named for the model is the model
    src <- if (length(list.files(dd, "parquet$")) <= 200) sprintf("(SELECT cell_id, val FROM read_parquet('%s/*.parquet') WHERE mdl_key = '%s')", dd, o$mdl_key)
           else sprintf("read_parquet('%s/%s.parquet')", dd, id)
    sprintf("SELECT cell_id, max(val) AS val, count(*) AS reps FROM %s GROUP BY cell_id", src)   # the merge consumes max(val) per cell
  } else if (o$grp == "merged") {
    dbExecute(con, sprintf("ATTACH '%s/merge.duckdb' AS m (READ_ONLY)", ad))
    id <- dbGetQuery(con, sprintf("SELECT mkey_id FROM m.mkey_map WHERE ms_merge_key = '%s'", o$mdl_key))$mkey_id
    if (length(id)) sprintf("SELECT cell_id, round_even(max(val), 0) AS val, 1 AS reps FROM read_parquet('%s/mc_parts/mkey_id=%d/*.parquet') GROUP BY cell_id", ad, id)
    else sprintf("SELECT cell_id, val, 1 AS reps FROM read_parquet('%s/dist_merged_global/**/*.parquet') WHERE mdl_key = '%s'", ad, o$mdl_key)
  } else stop("no row source for ", o$grp)
  q <- if (o$grp == "ax") "trunc(least(round_even(val, 0), 255))" else "trunc(least(val, 255))"
  dec <- native_raster_rows(terra::rast(file.path(ad, "native", o$rel)), grid)
  duckdb::duckdb_register(con, "dec", dec)
  r <- dbGetQuery(con, sprintf("
    WITH r AS (%s), x AS (SELECT cell_id, %s AS q FROM r WHERE %s >= 1),
         j AS (SELECT d.cell_id AS dc, x.cell_id AS xc, d.val AS dv, x.q AS xq FROM dec d FULL OUTER JOIN x ON CAST(d.cell_id AS BIGINT) = x.cell_id)
    SELECT (SELECT count(*) FROM dec) AS n_cog, (SELECT count(*) FROM x) AS n_rows,
           count(*) FILTER (WHERE xc IS NULL) AS only_cog, count(*) FILTER (WHERE dc IS NULL) AS only_rows,
           count(*) FILTER (WHERE dc IS NOT NULL AND xc IS NOT NULL AND dv <> xq) AS val_diff,
           (SELECT count(*) FROM r WHERE reps > 1) AS dup_cells,
           count(*) FILTER (WHERE xc IS NULL AND dv = 1) AS only_cog_at1, count(*) FILTER (WHERE dc IS NULL AND xq = 1) AS only_rows_at1,
           coalesce(max(abs(dv - xq)) FILTER (WHERE dc IS NOT NULL AND xc IS NOT NULL AND dv <> xq), 0) AS val_diff_max
    FROM j", rows_sql, q, q))
  tibble::tibble(ver = o$ver, grp = o$grp, file = o$file, etag = o$etag, mdl_key = o$mdl_key, r)
}

4.1 3a · Is a gross failure a shifted surface? (positional mid)

publish_native.qmd addresses each AquaMaps model by mid = seq_along(sorted served keys) and persists the map beside the partitions (_am_parts.map.csv). If the served set changed after the partitions were built, mid moved and a COG was painted from a neighbour’s cells. For each gross am failure: (1) does its COG decode to another model’s CURRENT surface (pixel hash equal), and at what offset in the sorted key order; (2) if not exactly, is a neighbour within ±10 positions the same size (pixel count), i.e. the neighbour’s rows moved since?

frac_worker <- function(o, cells, gpkg, g7) {
  suppressMessages({library(sf); library(msens)})
  p <- st_read(gpkg, quiet = TRUE, query = sprintf("SELECT * FROM rng_iucn WHERE mdl_key = 'rng_iucn|%s'", o$id_no)) |> st_zm(drop = TRUE)
  cl <- cells[cells$mdl_seq == o$mdl_seq, ]; ll <- cell_lonlat(cl$cell_id, g7, wrap = TRUE)
  pts <- st_as_sf(ll, coords = c("lon", "lat"), crs = 4326)
  near <- function(s2) {                                     # within 25 km: spherical; the planar fallback uses 0.25 deg
    sf_use_s2(s2); on.exit(sf_use_s2(TRUE))
    poly <- st_union(if (s2) st_make_valid(st_geometry(p)) else st_geometry(p)); st_crs(poly) <- 4326
    mean(lengths(st_is_within_distance(pts, poly, dist = if (s2) 25000 else 0.25)) > 0)
  }
  # s2 rejects a range with a degenerate loop (duplicate vertex): fall back to planar rather than call it a disagreement
  tryCatch(near(TRUE), error = function(e) tryCatch(near(FALSE), error = function(e2) NA_real_))
}

## 1 · The key is the hash of the PIXELS the file holds

M1 keyed the raster families by the hash of the raw source rows. That cannot tell `trunc` from `round`
(v8's 192 suitability-only merged models were painted with `trunc(max)`, v9's with `round(max)`: the same rows,
different files, one key) and gives two models whose rows differ below the quantum different keys for identical
pixels. `msens::pixel_hashes()` (0.47.0) hashes the values **as the writer stores them** — `trunc` into INT1U
(verified: every pixel of an AquaMaps model equals `trunc(val)`; 54% equal `round(val)`), sub-1 pixels dropped as
NoData — which equals `content_hashes_df(native_raster_rows(<the painted COG>))`. The check "does this object hold
the rows its key says" is then one equality, and the key is still known before any file is built.

## 3 · Why the failures fail

For every inconsistent object: the cell-level difference against the current rows (cells only in the COG, cells only in
the rows, shared cells with another value), whether the rows repeat a cell, and whether the COG is **another model's**
current surface (the positional-`mid` bug in `publish_native.qmd` painted 1,054 COGs from another species' partition).
A *small edge* failure differs in < 1% of its pixels; a *gross* one in ≥ 1%.

```{r}
#| label: diagnose
#| results: asis

fail <- ob |> filter(status == "inconsistent")
cat(sprintf("\n%d inconsistent objects (%d distinct by folder+name+bytes)\n\n", nrow(fail), nrow(distinct(fail, grp, file, etag))))

# the rows that SHOULD be in the object, per folder -- a self-contained worker: DuckDB reads the model's own rows (max(val) per cell,
# the value the merge consumes), the decoded COG is registered as a table, and ONE full outer join counts the differences
# (the base-R merge() version took 2-68 s per object; this takes 0.2-2 s and returns identical numbers, checked on 8 objects)
diag_worker <- function(o, atlas_root, grid) {
  suppressMessages({library(DBI); library(duckdb); library(terra); library(msens)})
  con <- dbConnect(duckdb(), config = list(threads = "2", memory_limit = "3GB")); on.exit(dbDisconnect(con, shutdown = TRUE))
  ad <- file.path(atlas_root, o$ver, "marine-atlas")
  rows_sql <- if (o$grp %in% c("am", "ax", "dps_nmfs", "vec_grid")) {
    ds <- sub("\\|.*$", "", o$mdl_key); id <- sub("^[^|]*\\|", "", o$mdl_key)
    dd <- file.path(ad, "dist", paste0("dataset=", ds))
    # a model can span several files (rng_turtle_swot_dps: 14 files, 6 models; vec_grid painted ONE file per model): for a small
    # dataset read the model's rows across ALL of them; for the big per-model ones the file named for the model is the model
    src <- if (length(list.files(dd, "parquet$")) <= 200) sprintf("(SELECT cell_id, val FROM read_parquet('%s/*.parquet') WHERE mdl_key = '%s')", dd, o$mdl_key)
           else sprintf("read_parquet('%s/%s.parquet')", dd, id)
    sprintf("SELECT cell_id, max(val) AS val, count(*) AS reps FROM %s GROUP BY cell_id", src)   # the merge consumes max(val) per cell
  } else if (o$grp == "merged") {
    dbExecute(con, sprintf("ATTACH '%s/merge.duckdb' AS m (READ_ONLY)", ad))
    id <- dbGetQuery(con, sprintf("SELECT mkey_id FROM m.mkey_map WHERE ms_merge_key = '%s'", o$mdl_key))$mkey_id
    if (length(id)) sprintf("SELECT cell_id, round_even(max(val), 0) AS val, 1 AS reps FROM read_parquet('%s/mc_parts/mkey_id=%d/*.parquet') GROUP BY cell_id", ad, id)
    else sprintf("SELECT cell_id, val, 1 AS reps FROM read_parquet('%s/dist_merged_global/**/*.parquet') WHERE mdl_key = '%s'", ad, o$mdl_key)
  } else stop("no row source for ", o$grp)
  q <- if (o$grp == "ax") "trunc(least(round_even(val, 0), 255))" else "trunc(least(val, 255))"
  dec <- native_raster_rows(terra::rast(file.path(ad, "native", o$rel)), grid)
  duckdb::duckdb_register(con, "dec", dec)
  r <- dbGetQuery(con, sprintf("
    WITH r AS (%s), x AS (SELECT cell_id, %s AS q FROM r WHERE %s >= 1),
         j AS (SELECT d.cell_id AS dc, x.cell_id AS xc, d.val AS dv, x.q AS xq FROM dec d FULL OUTER JOIN x ON CAST(d.cell_id AS BIGINT) = x.cell_id)
    SELECT (SELECT count(*) FROM dec) AS n_cog, (SELECT count(*) FROM x) AS n_rows,
           count(*) FILTER (WHERE xc IS NULL) AS only_cog, count(*) FILTER (WHERE dc IS NULL) AS only_rows,
           count(*) FILTER (WHERE dc IS NOT NULL AND xc IS NOT NULL AND dv <> xq) AS val_diff,
           (SELECT count(*) FROM r WHERE reps > 1) AS dup_cells,
           count(*) FILTER (WHERE xc IS NULL AND dv = 1) AS only_cog_at1, count(*) FILTER (WHERE dc IS NULL AND xq = 1) AS only_rows_at1,
           coalesce(max(abs(dv - xq)) FILTER (WHERE dc IS NOT NULL AND xc IS NOT NULL AND dv <> xq), 0) AS val_diff_max
    FROM j", rows_sql, q, q))
  tibble::tibble(ver = o$ver, grp = o$grp, file = o$file, etag = o$etag, mdl_key = o$mdl_key, r)
}
```

### 3a · Is a gross failure a shifted surface? (positional `mid`)

`publish_native.qmd` addresses each AquaMaps model by `mid = seq_along(sorted served keys)` and persists the map beside
the partitions (`_am_parts.map.csv`). If the served set changed after the partitions were built, `mid` moved and a COG was
painted from a neighbour's cells. For each gross `am` failure: (1) does its COG decode to another model's CURRENT surface
(pixel hash equal), and at what offset in the sorted key order; (2) if not exactly, is a neighbour within ±10 positions the
same size (pixel count), i.e. the neighbour's rows moved since?

## 4 · Repaint locally and verify

Every store key that has **no consistent object** is painted from the current rows with the family's encoding
(`publish_cog`: INT1U, nodata 0, DEFLATE, overviews) into `staging/`, decoded again and compared with the
pixel hash the key promises. A cell the rows repeat takes `max(val)` — what `turtle_sql()`/`merge_sql()` consume
(`er`/`suit` CTEs: `max(val) … GROUP BY ms_merge_key, cell_id`); the repeated rows are a defect of the
`ingest_turtles-swot-dps.qmd` ingest to fix in v10, not a rule the store should inherit.


## 5 · `rng_iucn` on v7: the 34 bbox failures, tested cell by cell

M1 compared the v7 surface's *extent* with the v9 PMTiles *header bounds*. Both are envelopes: a range crossing the
antimeridian has header bounds clipped to ±180 (e.g. −179 … 173), and a near-global deep-sea range has an envelope that
says nothing. The question that matters is whether the cells v7 actually draws lie on the range. So for each of the 34:
sample up to 300 of its v7 cells (`model_cell`, `usa05`, 0–360° longitude, wrapped to a point) and measure how many lie
within 25 km of the SOURCE range polygon (the indexed IUCN GeoPackage). **Agree = at least 95% of the sampled cells.**
The same test is run on 60 of the 1,426 that already passed, to show the criterion does not reject a good match.

```{r}
#| label: v7-34

frac_worker <- function(o, cells, gpkg, g7) {
  suppressMessages({library(sf); library(msens)})
  p <- st_read(gpkg, quiet = TRUE, query = sprintf("SELECT * FROM rng_iucn WHERE mdl_key = 'rng_iucn|%s'", o$id_no)) |> st_zm(drop = TRUE)
  cl <- cells[cells$mdl_seq == o$mdl_seq, ]; ll <- cell_lonlat(cl$cell_id, g7, wrap = TRUE)
  pts <- st_as_sf(ll, coords = c("lon", "lat"), crs = 4326)
  near <- function(s2) {                                     # within 25 km: spherical; the planar fallback uses 0.25 deg
    sf_use_s2(s2); on.exit(sf_use_s2(TRUE))
    poly <- st_union(if (s2) st_make_valid(st_geometry(p)) else st_geometry(p)); st_crs(poly) <- 4326
    mean(lengths(st_is_within_distance(pts, poly, dist = if (s2) 25000 else 0.25)) > 0)
  }
  # s2 rejects a range with a degenerate loop (duplicate vertex): fall back to planar rather than call it a disagreement
  tryCatch(near(TRUE), error = function(e) tryCatch(near(FALSE), error = function(e2) NA_real_))
}

    dbDisconnect(con7, shutdown = TRUE)
    chk$frac_cells_on_range[na_i] <- vapply(split(chk[na_i, ], seq_along(na_i)), frac_worker, 0, cells = cells, gpkg = as.character(iucn_gpkg), g7 = g7)

if (test_n > 0) chk <- slice_head(chk, n = test_n * 5)
```
